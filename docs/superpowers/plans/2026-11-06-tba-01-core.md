```python
"""Strict base64url without padding."""
import base64
import re

_ALPHABET = re.compile(r"[A-Za-z0-9_-]*")


def b64url_encode(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def b64url_decode(s: str) -> bytes:
    if not isinstance(s, str) or not _ALPHABET.fullmatch(s) or len(s) % 4 == 1:
        raise ValueError("invalid base64url")
    return base64.urlsafe_b64decode(s + "=" * (-len(s) % 4))
```

```python
"""Canonical transaction encoding shared by server, phone and evaluation.

The same rules are implemented in authenticator/js/canonical.js and are
cross-checked through shared/vectors.json.
"""
import json
import re

from .b64 import b64url_decode, b64url_encode

FIELDS = ("amount", "currency", "exp", "from", "nonce", "to", "to_name", "tx_id", "v")
MAX_INT = 2**53 - 1
HMAC_TAG = b"TBA-HMAC-v1"
SIG_TAG = b"TBA-SIG-v1"
QR_PREFIX = "tba1:"

_PATTERNS = {
    "from": re.compile(r"[0-9]{6,20}"),
    "to": re.compile(r"[0-9]{6,20}"),
    "to_name": re.compile(r"[A-Z0-9 ]{1,64}"),
    "nonce": re.compile(r"[A-Za-z0-9_-]{22}"),
    "tx_id": re.compile(r"[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}"),
    "currency": re.compile(r"[A-Z]{3}"),
}


class EncodingError(ValueError):
    """Raised when a transaction or payload is not in canonical form."""


def _is_int(value) -> bool:
    return type(value) is int  # rejects bool, which is a subclass of int


def validate_tx(tx: dict) -> None:
    if not isinstance(tx, dict):
        raise EncodingError("transaction must be an object")
    if set(tx) != set(FIELDS):
        raise EncodingError("transaction fields mismatch")
    if not _is_int(tx["v"]) or tx["v"] != 1:
        raise EncodingError("unsupported version")
    for key in ("amount", "exp"):
        if not _is_int(tx[key]) or not 1 <= tx[key] <= MAX_INT:
            raise EncodingError(f"invalid {key}")
    for key, pattern in _PATTERNS.items():
        value = tx[key]
        if not isinstance(value, str) or not pattern.fullmatch(value):
            raise EncodingError(f"invalid {key}")


def canonical_bytes(tx: dict) -> bytes:
    validate_tx(tx)
    return json.dumps(tx, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")


def _reject_float(text):
    raise EncodingError("floating point numbers are not allowed")


def _reject_constant(text):
    raise EncodingError("NaN/Infinity are not allowed")


def _no_duplicates(pairs):
    keys = [k for k, _ in pairs]
    if len(keys) != len(set(keys)):
        raise EncodingError("duplicate key")
    return dict(pairs)


def parse_canonical(data: bytes) -> dict:
    try:
        text = bytes(data).decode("ascii")
        obj = json.loads(
            text,
            object_pairs_hook=_no_duplicates,
            parse_float=_reject_float,
            parse_constant=_reject_constant,
        )
    except EncodingError:
        raise
    except (ValueError, TypeError, RecursionError) as exc:  # ValueError also covers over-long integers
        raise EncodingError("payload is not valid ASCII JSON") from exc
    if canonical_bytes(obj) != bytes(data):
        raise EncodingError("payload is not canonical")
    return obj


def tagged_message(tag: bytes, tx: dict) -> bytes:
    return tag + b"\x00" + canonical_bytes(tx)


def encode_qr_payload(tx: dict) -> str:
    return QR_PREFIX + b64url_encode(canonical_bytes(tx))


def decode_qr_payload(payload: str) -> dict:
    if not isinstance(payload, str) or not payload.startswith(QR_PREFIX):
        raise EncodingError("not a TBA transaction payload")
    try:
        data = b64url_decode(payload[len(QR_PREFIX):])
    except ValueError as exc:
        raise EncodingError("invalid base64url") from exc
    return parse_canonical(data)
```

```python
"""HOTP (RFC 4226) and TOTP (RFC 6238). This is the unbound baseline."""
import hashlib
import hmac
import re
import struct

_DIGITS = re.compile(r"[0-9]+")


def dynamic_truncate(mac: bytes, digits: int) -> str:
    if not 3 <= digits <= 9:
        raise ValueError("digits must be between 3 and 9")
    offset = mac[-1] & 0x0F
    code = struct.unpack(">I", mac[offset:offset + 4])[0] & 0x7FFFFFFF
    return str(code % 10**digits).zfill(digits)


def hotp(key: bytes, counter: int, digits: int = 6, algorithm: str = "sha1") -> str:
    mac = hmac.new(key, struct.pack(">Q", counter), getattr(hashlib, algorithm)).digest()
    return dynamic_truncate(mac, digits)


def totp(key: bytes, unix_time: int, digits: int = 6, step: int = 30) -> str:
    return hotp(key, unix_time // step, digits)


def _well_formed(code, digits: int) -> bool:
    return isinstance(code, str) and len(code) == digits and code.isascii() and bool(_DIGITS.fullmatch(code))


def totp_verify(key: bytes, code: str, unix_time: int, last_used_step: int | None,
                digits: int = 6, window: int = 1, step: int = 30) -> int | None:
    if not _well_formed(code, digits):
        return None
    current = unix_time // step
    for candidate in range(current - window, current + window + 1):
        if candidate < 0 or (last_used_step is not None and candidate <= last_used_step):
            continue
        if hmac.compare_digest(hotp(key, candidate, digits), code):
            return candidate
    return None
```

```python
"""Transaction-bound OTP: OCRA-style HMAC over the canonical transaction."""
import hashlib
import hmac
import re

from .encoding import HMAC_TAG, tagged_message
from .totp import dynamic_truncate

_DIGITS = re.compile(r"[0-9]+")


def tx_otp(key: bytes, tx: dict, digits: int = 8) -> str:
    mac = hmac.new(key, tagged_message(HMAC_TAG, tx), hashlib.sha256).digest()
    return dynamic_truncate(mac, digits)


def verify_tx_otp(key: bytes, tx: dict, code: str, digits: int = 8) -> bool:
    if not (isinstance(code, str) and len(code) == digits and code.isascii() and _DIGITS.fullmatch(code)):
        return False
    return hmac.compare_digest(tx_otp(key, tx, digits), code)
```

```python
"""Signature-bound confirmation. ECDSA P-256 is used by the phone (Web Crypto);
Ed25519 is only benchmarked.

Web Crypto produces raw r||s (IEEE P1363, 64 bytes); `cryptography` wants DER,
so we convert at this boundary.
"""
from cryptography.exceptions import InvalidSignature
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import ec, ed25519
from cryptography.hazmat.primitives.asymmetric.utils import decode_dss_signature, encode_dss_signature

from .encoding import SIG_TAG, tagged_message

P256_ORDER = 0xFFFFFFFF00000000FFFFFFFFFFFFFFFFBCE6FAADA7179E84F3B9CAC2FC632551
_ECDSA = ec.ECDSA(hashes.SHA256())


def generate_p256() -> ec.EllipticCurvePrivateKey:
    return ec.generate_private_key(ec.SECP256R1())


def public_key_to_raw(public_key: ec.EllipticCurvePublicKey) -> bytes:
    return public_key.public_bytes(serialization.Encoding.X962, serialization.PublicFormat.UncompressedPoint)


def public_key_from_raw(data: bytes) -> ec.EllipticCurvePublicKey:
    if not isinstance(data, bytes) or len(data) != 65 or data[0] != 0x04:
        raise ValueError("expected 65-byte uncompressed P-256 point")
    return ec.EllipticCurvePublicKey.from_encoded_point(ec.SECP256R1(), data)


def raw_to_der(signature: bytes) -> bytes:
    return encode_dss_signature(int.from_bytes(signature[:32], "big"), int.from_bytes(signature[32:], "big"))


def sign_tx(private_key: ec.EllipticCurvePrivateKey, tx: dict) -> bytes:
    r, s = decode_dss_signature(private_key.sign(tagged_message(SIG_TAG, tx), _ECDSA))
    return r.to_bytes(32, "big") + s.to_bytes(32, "big")


def verify_tx_signature(public_key: ec.EllipticCurvePublicKey, tx: dict, signature: bytes) -> bool:
    if not isinstance(signature, bytes) or len(signature) != 64:
        return False
    r = int.from_bytes(signature[:32], "big")
    s = int.from_bytes(signature[32:], "big")
    if not (0 < r < P256_ORDER and 0 < s < P256_ORDER):
        return False
    try:
        public_key.verify(encode_dss_signature(r, s), tagged_message(SIG_TAG, tx), _ECDSA)
        return True
    except InvalidSignature:
        return False


def generate_ed25519() -> ed25519.Ed25519PrivateKey:
    return ed25519.Ed25519PrivateKey.generate()


def sign_tx_ed25519(private_key: ed25519.Ed25519PrivateKey, tx: dict) -> bytes:
    return private_key.sign(tagged_message(SIG_TAG, tx))


def verify_tx_signature_ed25519(public_key: ed25519.Ed25519PublicKey, tx: dict, signature: bytes) -> bool:
    try:
        public_key.verify(signature, tagged_message(SIG_TAG, tx))
        return True
    except InvalidSignature:
        return False
```

```python
"""Fresh transactions: 128-bit nonce, random tx_id, short expiry."""
import secrets
import uuid

from .b64 import b64url_encode
from .encoding import validate_tx

NONCE_BYTES = 16
TTL_SECONDS = 120


def new_nonce() -> str:
    return b64url_encode(secrets.token_bytes(NONCE_BYTES))


def new_tx(from_account: str, to_account: str, to_name: str, amount: int, *,
           now: int, currency: str = "VND", ttl: int = TTL_SECONDS) -> dict:
    tx = {
        "v": 1,
        "tx_id": str(uuid.uuid4()),
        "from": from_account,
        "to": to_account,
        "to_name": to_name,
        "amount": amount,
        "currency": currency,
        "nonce": new_nonce(),
        "exp": now + ttl,
    }
    validate_tx(tx)
    return tx


def is_expired(tx: dict, now: int) -> bool:
    return now > tx["exp"]
```

```python
ECDSA signatures are randomized, so regenerating changes the "signature"
fields; tests only verify them, so any regenerated file is valid.
"""

import json
from pathlib import Path

from cryptography.hazmat.primitives.asymmetric import ec

from tba_core.encoding import canonical_bytes, encode_qr_payload
from tba_core.hmac_otp import tx_otp
from tba_core.signature import P256_ORDER, public_key_to_raw, sign_tx

OUT = Path(__file__).resolve().parents[1] / "shared" / "vectors.json"
HMAC_KEY = bytes(range(32))
TOTP_KEY = b"12345678901234567890"
_d = int.from_bytes(hashlib.sha256(b"tba-test-vector-key").digest(), "big") % (P256_ORDER - 1) + 1
SIGNING_KEY = ec.derive_private_key(_d, ec.SECP256R1())

TXS = [
    {"v": 1, "tx_id": "3f2b8c1e-5a4d-4e7f-9b6a-1c2d3e4f5a6b", "from": "0011223344",
     "to": "0123456789", "to_name": "TRAN THI B", "amount": 1000000, "currency": "VND",
     "nonce": b64url_encode(bytes(range(16))), "exp": 1800000120},
    {"v": 1, "tx_id": "a1b2c3d4-0000-4000-8000-000000000001", "from": "0011223344",
     "to": "0999888777", "to_name": "LE VAN M", "amount": 9000000, "currency": "VND",
     "nonce": b64url_encode(bytes(range(240, 256))), "exp": 1800000300},
    {"v": 1, "tx_id": "a1b2c3d4-0000-4000-8000-000000000002", "from": "0123456789",
     "to": "0011223344", "to_name": "NGUYEN VAN A", "amount": 2**53 - 1, "currency": "VND",
     "nonce": b64url_encode(b"\xff" * 16), "exp": 2**53 - 1},
]

TOTP = [(59, "94287082"), (1111111109, "07081804"), (1111111111, "14050471"),
        (1234567890, "89005924"), (2000000000, "69279037"), (20000000000, "65353130")]


def _p(text) -> str:
    data = text.encode("utf-8") if isinstance(text, str) else text
    return "tba1:" + b64url_encode(data)


def invalid_payloads() -> list[dict]:
    base = TXS[0]
    canon = canonical_bytes(base).decode()
    missing = {k: v for k, v in base.items() if k != "nonce"}
    cases = [
        ("whitespace", _p(json.dumps(base, sort_keys=True))),
        ("unsorted_keys", _p(json.dumps(base, separators=(",", ":")))),
        ("float_amount", _p(canon.replace('"amount":1000000', '"amount":1000000.0'))),
        ("duplicate_key", _p(canon.replace('{"amount":1000000', '{"amount":1,"amount":1000000'))),
        ("nfd_name_utf8", _p(canon.replace('"TRAN THI B"', '"TRA\u0300N THI B"'))),
        ("escaped_unicode_name", _p(canon.replace('"TRAN THI B"', '"TR\\u00c0N THI B"'))),
        ("extra_field", _p(canon[:-1] + ',"zz":1}')),
        ("missing_field", _p(json.dumps(missing, sort_keys=True, separators=(",", ":")))),
        ("bool_amount", _p(canon.replace('"amount":1000000', '"amount":true'))),
        ("negative_amount", _p(canon.replace('"amount":1000000', '"amount":-1000000'))),
        ("too_large_amount", _p(canon.replace('"amount":1000000', '"amount":9007199254740993'))),
        ("no_prefix", b64url_encode(canon.encode())),
        ("bad_base64", "tba1:!!!!"),
        ("trailing_newline", _p(canon + "\n")),
    ]
    return [{"name": n, "payload": p} for n, p in cases]


def main() -> None:
    vectors = {
        "hmac_key": b64url_encode(HMAC_KEY),
        "totp_key": b64url_encode(TOTP_KEY),
        "p256_public_key": b64url_encode(public_key_to_raw(SIGNING_KEY.public_key())),
        "transactions": [
            {
                "tx": tx,
                "canonical": canonical_bytes(tx).decode(),
                "qr_payload": encode_qr_payload(tx),
                "hmac_otp_6": tx_otp(HMAC_KEY, tx, 6),
                "hmac_otp_8": tx_otp(HMAC_KEY, tx, 8),
                "signature": b64url_encode(sign_tx(SIGNING_KEY, tx)),
            }
            for tx in TXS
        ],
        "totp": [{"t": t, "digits": 8, "code": c} for t, c in TOTP],
        "invalid_qr_payloads": invalid_payloads(),
    }
    OUT.parent.mkdir(exist_ok=True)
    OUT.write_text(json.dumps(vectors, indent=2) + "\n")
    print(f"wrote {OUT}")


if __name__ == "__main__":
    main()
```
