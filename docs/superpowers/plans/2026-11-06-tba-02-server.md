> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Server ngân hàng demo với 4 chế độ xác thực, đăng ký thiết bị, tạo giao dịch, xác minh nguyên tử và audit log.

**Architecture:** Logic tách khỏi HTTP: `auth.py`, `transfers.py`, `verify.py` nhận một `sqlite3.Connection` và trả về kết quả; `app.py` chỉ nối route với các hàm đó. Đồng hồ được tiêm vào (`clock`) để test điều khiển được thời gian.

**Spec:** `docs/superpowers/specs/2026-11-06-tba-midterm-design.md` (mục 4.3-4.6)

## Global Constraints

- Chế độ: `baseline`, `hmac`, `sig`, `sig_blind`.
- OTP `hmac` 8 chữ số, TOTP 6 chữ số; tối đa 3 lần thử; hạn giao dịch 120 s; hạn token đăng ký 300 s.
- Luôn xác minh trên `transactions.canonical`, không dùng dữ liệu giao dịch từ client.
- Mọi bước xác nhận nằm trong một `BEGIN IMMEDIATE`.
- `/api/device/confirm` không bao giờ trả 5xx.
- Giao diện tiếng Anh. Trang xác nhận và trang đăng ký có `<div id="qr" data-payload="...">` (test và addon MitB dựa vào thuộc tính này).

```python
import sqlite3
from contextlib import contextmanager

SCHEMA = """
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  pw_salt BLOB NOT NULL,
  pw_hash BLOB NOT NULL,
  account TEXT UNIQUE NOT NULL,
  display_name TEXT NOT NULL,
  balance INTEGER NOT NULL CHECK (balance >= 0)
);
CREATE TABLE IF NOT EXISTS sessions (
  token TEXT PRIMARY KEY,
  user_id INTEGER NOT NULL REFERENCES users(id),
  exp INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS enroll_tokens (
  token TEXT PRIMARY KEY,
  user_id INTEGER NOT NULL REFERENCES users(id),
  exp INTEGER NOT NULL,
  used INTEGER NOT NULL DEFAULT 0
);
CREATE TABLE IF NOT EXISTS devices (
  id TEXT PRIMARY KEY,
  user_id INTEGER UNIQUE NOT NULL REFERENCES users(id),
  public_key BLOB NOT NULL,
  k_totp BLOB NOT NULL,
  k_hmac BLOB NOT NULL,
  last_totp_step INTEGER,
  created_at INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS transactions (
  tx_id TEXT PRIMARY KEY,
  user_id INTEGER NOT NULL REFERENCES users(id),
  mode TEXT NOT NULL,
  canonical BLOB NOT NULL,
  state TEXT NOT NULL CHECK (state IN ('pending', 'executed', 'failed', 'expired')),
  attempts INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL,
  evidence BLOB
);
CREATE TABLE IF NOT EXISTS audit_log (
  id INTEGER PRIMARY KEY,
  ts INTEGER NOT NULL,
  user_id INTEGER,
  tx_id TEXT,
  event TEXT NOT NULL,
  detail TEXT
);
"""


def connect(path: str) -> sqlite3.Connection:
    # Autocommit mode; multi-statement changes use write_tx().
    conn = sqlite3.connect(path, isolation_level=None, timeout=30, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_db(path: str) -> None:
    conn = connect(path)
    try:
        conn.executescript(SCHEMA)
    finally:
        conn.close()


@contextmanager
def write_tx(conn: sqlite3.Connection):
    """Serialize writers: BEGIN IMMEDIATE takes the write lock up front."""
    conn.execute("BEGIN IMMEDIATE")
    try:
        yield conn
    except BaseException:
        conn.execute("ROLLBACK")
        raise
    else:
        conn.execute("COMMIT")


def audit(conn, ts: int, user_id, tx_id, event: str, detail: str = "") -> None:
    conn.execute(
        "INSERT INTO audit_log (ts, user_id, tx_id, event, detail) VALUES (?, ?, ?, ?, ?)",
        (ts, user_id, tx_id, event, detail),
    )
```

```python
import hashlib
import hmac
import secrets
import uuid
from dataclasses import dataclass

from .db import audit, write_tx

_SCRYPT = {"n": 2**14, "r": 8, "p": 1, "dklen": 32}


@dataclass(frozen=True)
class DemoUser:
    username: str
    password: str
    account: str
    name: str
    balance: int


DEMO_USERS = (
    DemoUser("alice", "alice-pass", "0011223344", "NGUYEN VAN A", 100_000_000),
    DemoUser("bob", "bob-pass", "0123456789", "TRAN THI B", 5_000_000),
    DemoUser("mallory", "mallory-pass", "0999888777", "LE VAN M", 0),
)


def hash_password(password: str, salt: bytes | None = None) -> tuple[bytes, bytes]:
    salt = secrets.token_bytes(16) if salt is None else salt
    return salt, hashlib.scrypt(password.encode("utf-8"), salt=salt, **_SCRYPT)


def seed_demo_users(conn) -> None:
    for u in DEMO_USERS:
        if conn.execute("SELECT 1 FROM users WHERE username = ?", (u.username,)).fetchone():
            continue
        salt, pw_hash = hash_password(u.password)
        conn.execute(
            "INSERT INTO users (username, pw_salt, pw_hash, account, display_name, balance) VALUES (?, ?, ?, ?, ?, ?)",
            (u.username, salt, pw_hash, u.account, u.name, u.balance),
        )


def authenticate(conn, username: str, password: str):
    row = conn.execute("SELECT * FROM users WHERE username = ?", (username,)).fetchone()
    if row is None:
        hash_password(password)  # spend the same time as a real check
        return None
    _, candidate = hash_password(password, bytes(row["pw_salt"]))
    return row if hmac.compare_digest(candidate, bytes(row["pw_hash"])) else None


def create_session(conn, user_id: int, now: int, ttl: int) -> str:
    token = secrets.token_urlsafe(32)
    conn.execute("INSERT INTO sessions (token, user_id, exp) VALUES (?, ?, ?)", (token, user_id, now + ttl))
    return token


def user_for_session(conn, token: str, now: int):
    return conn.execute(
        "SELECT users.* FROM sessions JOIN users ON users.id = sessions.user_id "
        "WHERE sessions.token = ? AND sessions.exp >= ?",
        (token, now),
    ).fetchone()


def delete_session(conn, token: str | None) -> None:
    if token:
        conn.execute("DELETE FROM sessions WHERE token = ?", (token,))


def create_enroll_token(conn, user_id: int, now: int, ttl: int) -> str:
    token = secrets.token_urlsafe(24)
    conn.execute("INSERT INTO enroll_tokens (token, user_id, exp) VALUES (?, ?, ?)", (token, user_id, now + ttl))
    return token


def enroll_device(conn, token: str, public_key_raw: bytes, now: int) -> tuple[str, bytes, bytes] | None:
    with write_tx(conn):
        row = conn.execute("SELECT * FROM enroll_tokens WHERE token = ?", (token,)).fetchone()
        if row is None or row["used"] or now > row["exp"]:
            audit(conn, now, row["user_id"] if row else None, None, "enroll_rejected", "invalid token")
            return None
        conn.execute("UPDATE enroll_tokens SET used = 1 WHERE token = ?", (token,))
        device_id = str(uuid.uuid4())
        k_totp, k_hmac = secrets.token_bytes(20), secrets.token_bytes(32)
        conn.execute("DELETE FROM devices WHERE user_id = ?", (row["user_id"],))
        conn.execute(
            "INSERT INTO devices (id, user_id, public_key, k_totp, k_hmac, last_totp_step, created_at) "
            "VALUES (?, ?, ?, ?, ?, NULL, ?)",
            (device_id, row["user_id"], public_key_raw, k_totp, k_hmac, now),
        )
        audit(conn, now, row["user_id"], None, "device_enrolled", device_id)
        return device_id, k_totp, k_hmac


def device_for_user(conn, user_id: int):
    return conn.execute("SELECT * FROM devices WHERE user_id = ?", (user_id,)).fetchone()
```

```python
import re

from tba_core.challenge import new_tx
from tba_core.encoding import canonical_bytes, parse_canonical

from .db import audit, write_tx

_ACCOUNT = re.compile(r"[0-9]{6,20}")
_AMOUNT = re.compile(r"[0-9]{1,16}")


class TransferError(Exception):
    """User-facing validation error."""


def lookup_account(conn, account: str):
    if not isinstance(account, str) or not _ACCOUNT.fullmatch(account):
        return None
    return conn.execute("SELECT account, display_name FROM users WHERE account = ?", (account,)).fetchone()


def create_transfer(conn, settings, user, to: str, amount_text: str, now: int) -> dict:
    if not _ACCOUNT.fullmatch(to):
        raise TransferError("Invalid account number")
    if not _AMOUNT.fullmatch(amount_text) or int(amount_text) < 1:
        raise TransferError("Invalid amount")
    amount = int(amount_text)
    if to == user["account"]:
        raise TransferError("Cannot transfer to your own account")
    recipient = lookup_account(conn, to)
    if recipient is None:
        raise TransferError("Unknown recipient account")
    with write_tx(conn):
        if conn.execute("SELECT 1 FROM devices WHERE user_id = ?", (user["id"],)).fetchone() is None:
            raise TransferError("Enroll an authenticator first")
        balance = conn.execute("SELECT balance FROM users WHERE id = ?", (user["id"],)).fetchone()["balance"]
        if amount > balance:
            raise TransferError("Insufficient balance")
        tx = new_tx(user["account"], to, recipient["display_name"], amount, now=now, ttl=settings.tx_ttl)
        conn.execute(
            "INSERT INTO transactions (tx_id, user_id, mode, canonical, state, attempts, created_at) "
            "VALUES (?, ?, ?, ?, 'pending', 0, ?)",
            (tx["tx_id"], user["id"], settings.mode, canonical_bytes(tx), now),
        )
        audit(conn, now, user["id"], tx["tx_id"], "tx_created", f"to={to} amount={amount}")
    return tx


def get_tx_for_user(conn, tx_id: str, user_id: int) -> dict | None:
    row = conn.execute(
        "SELECT canonical FROM transactions WHERE tx_id = ? AND user_id = ?", (tx_id, user_id)
    ).fetchone()
    return parse_canonical(bytes(row["canonical"])) if row else None


def current_state(conn, tx_id: str, user_id: int, now: int) -> str | None:
    row = conn.execute(
        "SELECT state, canonical FROM transactions WHERE tx_id = ? AND user_id = ?", (tx_id, user_id)
    ).fetchone()
    if row is None:
        return None
    if row["state"] == "pending" and now > parse_canonical(bytes(row["canonical"]))["exp"]:
        conn.execute("UPDATE transactions SET state = 'expired' WHERE tx_id = ? AND state = 'pending'", (tx_id,))
        return "expired"
    return row["state"]
```

```python
import pytest

from server import auth, db, transfers
from server.config import Settings
from server.verify import Result, confirm_otp, confirm_signature
from tba_core.hmac_otp import tx_otp
from tba_core.signature import generate_p256, public_key_to_raw, sign_tx
from tba_core.totp import totp

NOW = 1_800_000_000


def setup(tmp_path, mode):
    path = str(tmp_path / f"{mode}.sqlite3")
    db.init_db(path)
    conn = db.connect(path)
    auth.seed_demo_users(conn)
    alice = auth.authenticate(conn, "alice", "alice-pass")
    sk = generate_p256()
    token = auth.create_enroll_token(conn, alice["id"], NOW, 300)
    device_id, k_totp, k_hmac = auth.enroll_device(conn, token, public_key_to_raw(sk.public_key()), NOW)
    settings = Settings(mode=mode, db_path=path)
    tx = transfers.create_transfer(conn, settings, alice, "0123456789", "1000000", NOW)
    return conn, settings, alice, tx, dict(sk=sk, device_id=device_id, k_totp=k_totp, k_hmac=k_hmac)


def balance(conn, account):
    return conn.execute("SELECT balance FROM users WHERE account = ?", (account,)).fetchone()[0]


def state(conn, tx_id):
    return conn.execute("SELECT state, attempts FROM transactions WHERE tx_id = ?", (tx_id,)).fetchone()


def test_hmac_correct_code_executes(tmp_path):
    conn, s, alice, tx, d = setup(tmp_path, "hmac")
    assert confirm_otp(conn, s, alice["id"], tx["tx_id"], tx_otp(d["k_hmac"], tx), NOW + 5) is Result.OK
    assert state(conn, tx["tx_id"])["state"] == "executed"
    assert balance(conn, "0123456789") == 6_000_000
    assert balance(conn, "0011223344") == 99_000_000


def test_hmac_code_for_other_tx_rejected(tmp_path):
    conn, s, alice, tx, d = setup(tmp_path, "hmac")
    forged = dict(tx, to="0999888777", to_name="LE VAN M")
    assert confirm_otp(conn, s, alice["id"], tx["tx_id"], tx_otp(d["k_hmac"], forged), NOW) is Result.REJECTED
    assert state(conn, tx["tx_id"])["attempts"] == 1


def test_lockout_after_three_failures(tmp_path):
    conn, s, alice, tx, d = setup(tmp_path, "hmac")
    for _ in range(3):
```

```python
"""Confirmation of pending transactions.

Every confirmation runs inside one BEGIN IMMEDIATE transaction, so two
concurrent requests can never both observe state='pending' (scenario S8).
The transaction is always rebuilt from the server's own canonical copy.
"""
from enum import Enum
from typing import Callable

from tba_core.encoding import parse_canonical
from tba_core.hmac_otp import verify_tx_otp
from tba_core.signature import public_key_from_raw, verify_tx_signature
from tba_core.totp import totp_verify

from .db import audit, write_tx

SIG_MODES = ("sig", "sig_blind")


class Result(str, Enum):
    OK = "ok"
    REJECTED = "rejected"


Check = Callable[[dict, object], bytes | None]  # (tx, device_row) -> evidence or None


def _execute_transfer(conn, tx: dict) -> bool:
    debited = conn.execute(
        "UPDATE users SET balance = balance - ? WHERE account = ? AND balance >= ?",
        (tx["amount"], tx["from"], tx["amount"]),
    )
    if debited.rowcount != 1:
        return False
    credited = conn.execute("UPDATE users SET balance = balance + ? WHERE account = ?", (tx["amount"], tx["to"]))
    if credited.rowcount != 1:
        raise RuntimeError("recipient vanished")  # rolls back the debit
    return True


def _confirm(conn, settings, user_id: int, tx_id: str, now: int, check: Check) -> Result:
    with write_tx(conn):
        row = conn.execute("SELECT * FROM transactions WHERE tx_id = ?", (tx_id,)).fetchone()
        if row is None or row["user_id"] != user_id:
            audit(conn, now, user_id, tx_id, "confirm_rejected", "unknown transaction")
            return Result.REJECTED
        if row["state"] != "pending":
            audit(conn, now, user_id, tx_id, "confirm_rejected", f"state={row['state']}")
            return Result.REJECTED
        tx = parse_canonical(bytes(row["canonical"]))
        if now > tx["exp"]:
            conn.execute("UPDATE transactions SET state = 'expired' WHERE tx_id = ?", (tx_id,))
            audit(conn, now, user_id, tx_id, "confirm_rejected", "expired")
            return Result.REJECTED
        device = conn.execute("SELECT * FROM devices WHERE user_id = ?", (user_id,)).fetchone()
        evidence = check(tx, device) if device is not None else None
        if evidence is None:
            attempts = row["attempts"] + 1
            new_state = "failed" if attempts >= settings.max_attempts else "pending"
            conn.execute("UPDATE transactions SET attempts = ?, state = ? WHERE tx_id = ?", (attempts, new_state, tx_id))
            audit(conn, now, user_id, tx_id, "confirm_failed", f"attempt {attempts}, state={new_state}")
            return Result.REJECTED
        if not _execute_transfer(conn, tx):
            conn.execute("UPDATE transactions SET state = 'failed' WHERE tx_id = ?", (tx_id,))
            audit(conn, now, user_id, tx_id, "confirm_rejected", "insufficient funds")
            return Result.REJECTED
        conn.execute(
            "UPDATE transactions SET state = 'executed', evidence = ? WHERE tx_id = ? AND state = 'pending'",
            (evidence, tx_id),
        )
        audit(conn, now, user_id, tx_id, "executed", settings.mode)
        return Result.OK


def confirm_otp(conn, settings, user_id: int, tx_id: str, code: str, now: int) -> Result:
    def check(tx, device):
        if settings.mode == "baseline":
            step = totp_verify(bytes(device["k_totp"]), code, now, device["last_totp_step"], digits=settings.totp_digits)
            if step is None:
                return None
            conn.execute("UPDATE devices SET last_totp_step = ? WHERE id = ?", (step, device["id"]))
            return code.encode("ascii")
        if settings.mode == "hmac" and verify_tx_otp(bytes(device["k_hmac"]), tx, code, settings.otp_digits):
            return code.encode("ascii")
        return None

    return _confirm(conn, settings, user_id, tx_id, now, check)


def confirm_signature(conn, settings, device_id: str, tx_id: str, signature: bytes, now: int) -> Result:
    owner = conn.execute("SELECT user_id FROM devices WHERE id = ?", (device_id,)).fetchone()
    if owner is None:
        audit(conn, now, None, tx_id, "confirm_rejected", "unknown device")
        return Result.REJECTED

    def check(tx, device):
        if settings.mode not in SIG_MODES or device["id"] != device_id:
            return None
        public_key = public_key_from_raw(bytes(device["public_key"]))
        return signature if verify_tx_signature(public_key, tx, signature) else None

    return _confirm(conn, settings, owner["user_id"], tx_id, now, check)
```

```python
    settings = settings or settings_from_env()
    clock = clock or (lambda: int(time.time()))
    dbm.init_db(settings.db_path)
    seed_conn = dbm.connect(settings.db_path)
    try:
        auth.seed_demo_users(seed_conn)
    finally:

        if state is None:
            raise HTTPException(status_code=404)
        return {"state": state}

        if row is None:
            raise HTTPException(status_code=404)

            raw = b64url_decode(req.public_key)
            public_key_from_raw(raw)

            return JSONResponse({"error": "invalid_token"}, status_code=403)

        return {"device_id": device_id, "k_totp": b64url_encode(k_totp),
                "k_hmac": b64url_encode(k_hmac), "mode": settings.mode}

        except ValueError:
            return {"result": verify.Result.REJECTED.value}
        result = verify.confirm_signature(conn, settings, req.device_id, req.tx_id, signature, clock())
        return {"result": result.value}

    return app
```

```python
    """Return (app, browser, phone). Separate clients = separate cookie jars."""
    app = create_app(settings, clock=clock)
    return app, TestClient(app, base_url="https://testserver"), TestClient(app, base_url="https://testserver")
```

- [ ] **Step 1: Viết `scripts/dev_cert.sh`**

```bash
#!/usr/bin/env bash
# Create a locally trusted certificate for the demo server with mkcert.
# Usage: scripts/dev_cert.sh <extra-host-or-ip>...
set -euo pipefail
command -v mkcert >/dev/null || { echo "Install mkcert first: brew install mkcert nss"; exit 1; }
mkdir -p certs
mkcert -install
LAN_IP=$(ipconfig getifaddr en0 2>/dev/null || hostname -I 2>/dev/null | awk '{print $1}')
mkcert -cert-file certs/cert.pem -key-file certs/key.pem localhost 127.0.0.1 ${LAN_IP:+"$LAN_IP"} "$@"
echo "Root CA to install on the phone: $(mkcert -CAROOT)/rootCA.pem"
echo "Server URL for the phone: https://${LAN_IP:-localhost}:8443/app/"
```

Run: `chmod +x scripts/dev_cert.sh`

- [ ] **Step 2: Thêm target vào `Makefile`** (dòng lệnh thụt bằng TAB)

```make
.PHONY: cert run

```
